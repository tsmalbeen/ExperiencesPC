def recherche_dichotomique(t, x):
    gauche, droite = 0, len(t) - 1
    while gauche <= droite:
        milieu = (gauche + droite) // 2
        if t[milieu] == x:
            return milieu
        elif t[milieu] < x:
            gauche = milieu + 1
        else:
            droite = milieu - 1
    return -1

    Algorithme glouton
    À chaque étape, on rend la plus grande valeur qui ne dépasse pas ce qu'il reste à rendre.
    
        
    Meilleure solution possible
    Calculée en essayant toutes les combinaisons utiles (le moins de pièces possible).
    
      
    
      
    def rendu_glouton(somme, valeurs):     # valeurs triées par ordre décroissant
    rendu = []
    for v in valeurs:
        while somme >= v:
            rendu.append(v)
            somme = somme - v
    return rendu
