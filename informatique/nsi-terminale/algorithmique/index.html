def tri_fusion(t):
    if len(t) <= 1:
        return t                      # cas de base : déjà trié
    milieu = len(t) // 2
    gauche = tri_fusion(t[:milieu])   # diviser...
    droite = tri_fusion(t[milieu:])
    return fusion(gauche, droite)     # ...puis combiner

def fusion(a, b):
    resultat, i, j = [], 0, 0
    while i < len(a) and j < len(b):
        if a[i] <= b[j]:
            resultat.append(a[i]); i += 1
        else:
            resultat.append(b[j]); j += 1
    return resultat + a[i:] + b[j:]

La case s contient le nombre minimal de pièces pour rendre s. Elle se calcule à partir des cases déjà remplies : pour chaque pièce p, on essaie 1 + case(s − p) et on garde le minimum.

      

      

      

        
def rendu_dynamique(somme, pieces):
    nb = [0] + [float('inf')] * somme
    for s in range(1, somme + 1):
        for p in pieces:
            if p <= s and nb[s - p] + 1 < nb[s]:
                nb[s] = nb[s - p] + 1
    return nb[somme]
